export const validarCoordenadas = (req, res, next) => {
  const { latitud, longitud } = req.body

  if (
    latitud && longitud &&
    parseFloat(latitud) >= -90 && parseFloat(latitud) <= 90 &&
    parseFloat(longitud) >= -180 && parseFloat(longitud) <= 180
  ) {
    next()
  } else {
    res.status(400).json({ message: 'Coordenadas inválidas' })
  }
}
